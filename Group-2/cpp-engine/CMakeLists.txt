cmake_minimum_required(VERSION 3.16)
project(DisasterRescueCommander LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(engine src/Game.cpp src/Entities.cpp src/Dijkstra.cpp src/GameManager.cpp src/Scenario.cpp)
target_include_directories(engine PUBLIC include vendor)
if(EMSCRIPTEN)
  add_executable(driving src/driving-main.cpp)
  target_include_directories(driving PRIVATE include vendor)
  target_compile_options(driving PRIVATE -fexceptions)
  target_link_options(driving PRIVATE --bind -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createDrivingRewards -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web,node -sFILESYSTEM=0 --no-entry)
  set_target_properties(driving PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/../frontend/public/wasm")
  add_executable(game src/main.cpp)
  target_link_libraries(game PRIVATE engine)
  target_compile_options(engine PRIVATE -fexceptions)
  target_compile_options(game PRIVATE -fexceptions)
  target_link_options(game PRIVATE --bind -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sEXPORTED_RUNTIME_METHODS=getExceptionMessage,decrementExceptionRefcount -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createRescueEngine -sALLOW_MEMORY_GROWTH=1 -sENVIRONMENT=web,node --no-entry)
  set_target_properties(game PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/../frontend/public/wasm")
else()
  enable_testing()
  add_executable(driving_rewards_tests tests/driving_rewards_tests.cpp)
  target_include_directories(driving_rewards_tests PRIVATE include vendor)
  add_test(NAME driving_rewards_tests COMMAND driving_rewards_tests)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE engine)
  add_test(NAME engine_tests COMMAND engine_tests)
endif()
